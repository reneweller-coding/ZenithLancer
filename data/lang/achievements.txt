# Achievements and the statistics page (src/Achieve.h; check with ZenithLancer --achievements). The file lives in
# lang/ because the Catalog skips that folder: these blocks are not the Catalog's, and --validate stays quiet.
#
# achievement <name>                 a name: lower case ASCII, digits and _
#   title "..."  title_en "..."      its name, German and English (both needed)
#   desc "..."   desc_en "..."       what to do, German and English (both needed; no "=" in a text)
#   need <counter> <value>           reached when the counter gets to the value or above ...
#   need <counter> <value> lower=1   ... or to the value or below (best times); a counter never set reaches nothing
#   hidden 1                         its title and text stay secret until it is reached
#
# stat <counter>                     a line of the statistics page (in this order)
#   title "..."  title_en "..."
#   show count|time|clock|credits    a number; seconds as h:mm; seconds as m:ss ("–" when never set); credits
#
# The counters the game feeds (when each is counted: counterKeys() in src/Achieve.h): kills bosses boss_nohit
# bossretries missions nodamage allkills deaths seconds credits spent max_credits max_front cubes plans secrets
# episode1..episode6 campaign_diff (0 Kadett .. 6 Zenit) campaign_coop coop_missions arcade_runs arcade2_runs
# arcade_clean bossrush_done bossrush_best survival_seconds survival_daily ngplus practice replays_watched
# max_chain grazes school weekly seen heard gold challenges
